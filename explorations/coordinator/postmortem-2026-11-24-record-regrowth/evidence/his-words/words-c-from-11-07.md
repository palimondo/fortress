import json,sys
src=sys.argv[1];out=sys.argv[2]
n=0
with open(src,errors='replace') as f, open(out,'w') as o:
  for i,l in enumerate(f,1):
    if '"type":"user"' not in l[:400] and '"type": "user"' not in l[:400]:

    try: d=json.loads(l)

    if d.get('type')!='user': continue

    if t is None or t.lstrip().startswith('<'): continue
    n+=1
    o.write(f"=== #{n} line {i} {d.get('timestamp')}\n{t}\n\n")
