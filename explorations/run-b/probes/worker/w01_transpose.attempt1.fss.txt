component w01_transpose
export Executable

opr (x: Array[\RR64,ZZ32\])^T : Array[\RR64,(ZZ32,ZZ32)\] = do
    n = |x|
    r: Array[\RR64,(ZZ32,ZZ32)\] = array[\RR64\](1, n)
    for j <- 0#n do
        r[(0, j)] := x[j]
    end
    r
  end

opr (m: Array[\RR64,(ZZ32,ZZ32)\])^T : Array[\RR64,(ZZ32,ZZ32)\] = do
    (nr, nc) = |m.bounds|
    r: Array[\RR64,(ZZ32,ZZ32)\] = array[\RR64\](nc, nr)
    for (i, j) <- m.indices do
        r[(j, i)] := m[(i, j)]
    end
    r
  end

run(): () = do
    x: Array[\RR64,ZZ32\] = array[\RR64\](3)
    x[0] := 1.0
    x[1] := 2.0
    x[2] := 3.0
    m: Array[\RR64,(ZZ32,ZZ32)\] = array[\RR64\](2, 3)
    m[(0,0)] := 1.0
    m[(0,1)] := 2.0
    m[(0,2)] := 3.0
    m[(1,0)] := 4.0
    m[(1,1)] := 5.0
    m[(1,2)] := 6.0
    println("x   = " x)
    println("x^T = " (x^T))
    println("M   = " m)
    println("M^T = " (m^T))
  end

end
