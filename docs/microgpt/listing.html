1(* MicroGptFlat -- microGPT in native Fortress arrays, data designed after Hsu
2   (Run C4, the synthesis of rounds two and three: explorations/run-c4/design.md).
3   The model only: hyperparameters, the layout table, the four vector
4   functions, the step, Adam and the training loop, line for line against
5   explorations/apl/reference/hsu-flat/microgpt_concise.dyalog and the formulas
6   (the tour: explorations/run-c4/tour.md).  The array vocabulary is FlatArrays,
7   the loaders and the corpus FlatData.  Run from explorations/run-c4/src: the
8   paths are relative. *)
9component MicroGptFlat
10export MicroGptFlat
11export Executable
12import FlatArrays.{...}
13import FlatData.{...}
14import System.{...}
15
16(* --------------------------------------------------- hyperparameters ---- *)
17(nEmbd, blockSize, nHead, headDim, vocabSize, bosId) = (16, 16, 4, 4, 27, 26)
18(epsilon, lr0, beta1, beta2, epsilon_A, nSteps) = (10.0^(-5), 0.01, 0.85, 0.99, 10.0^(-8), 1000)
19(corpusPath, weightsDir) = ("../../apl/reference/dzaima/docs.txt", "../../apl/reference/dzaima/w")
20
21(* ^T on a matrix and on every plane of a rank-3 array, declared here and not
22   in the vocabulary because an API cannot declare a postfix operator (row 133) *)
23opr (m: Matrix[\RR64,r,c\])^T[\nat r, nat c\]: Matrix[\RR64,c,r\] = transpose(m)
24opr (t: Array3[\RR64,0,a,0,b,0,c\])^T[\nat a, nat b, nat c\]: Array3[\RR64,0,a,0,c,0,b\] = transpose(t)
25
26(* -------------------------------------- rmsn, sm and their backward ------ *)
27rmsn(x: Array[\RR64,ZZ32\]): Array[\RR64,ZZ32\] = x / SQRT (epsilon + (x DOT x) / |x|)
28sm(z: Array[\RR64,ZZ32\]): Array[\RR64,ZZ32\] = do e = exp(z - (BIG MAX z)); e / (SUM e) end
29mask: Array[\RR64,(ZZ32,ZZ32)\] = mat(blockSize, blockSize, fn (i: ZZ32, j: ZZ32): RR64 => if j > i then -(10.0^10) else 0.0 end)
30rmsn_b(dy: Array[\RR64,ZZ32\], x: Array[\RR64,ZZ32\]): Array[\RR64,ZZ32\] = do
31    r = SQRT (epsilon + (x DOT x) / |x|); y = x / r
32    (dy - y ((y DOT dy) / |y|)) / r
33  end
34sm_b(p: Array[\RR64,ZZ32\], dy: Array[\RR64,ZZ32\]): Array[\RR64,ZZ32\] = p × (dy - (p DOT dy))
35
36(* ------------------------------- tactic 2/8: one vector, nine views ------ *)
37matName(i: ZZ32): String =
38    if i = 0 then "wte" elif i = 1 then "wpe" elif i = 2 then "lm_head" elif i = 3 then "attn_wq" elif i = 4 then "attn_wk"
39    elif i = 5 then "attn_wv" elif i = 6 then "attn_wo" elif i = 7 then "mlp_fc1" else "mlp_fc2" end
40matShape(i: ZZ32): (ZZ32, ZZ32) =
41    if (i = 0) OR (i = 2) then (vocabSize, nEmbd) elif i = 7 then (4 nEmbd, nEmbd) elif i = 8 then (nEmbd, 4 nEmbd) else (nEmbd, nEmbd) end
42matCount(i: ZZ32): ZZ32 = do (r, c) = matShape(i); r c end
43matOffset(i: ZZ32): ZZ32 = SUM[\ZZ32\][j <- 0#i] matCount(j)
44nParams(): ZZ32 = matOffset(9)
45view(p: Array[\RR64,ZZ32\], i: ZZ32): Array[\RR64,(ZZ32,ZZ32)\] = do (r, c) = matShape(i); view(p, matOffset(i), r, c) end
46
47(* -------------------- tactic 3/5: the corpus as one padded integer matrix - *)
48corpus: Corpus = loadCorpus(corpusPath, blockSize, bosId)
49
50(* ------------------------- the step: keys -> (loss, flat gradient). Pure. -- *)
51step(p: Array[\RR64,ZZ32\], b: Array[\ZZ32,ZZ32\]): (RR64, Array[\RR64,ZZ32\]) = do
52    ids = corpus.tokens(b, 0); tg = corpus.tokens(b, 1); vm = corpus.valid(b); nv = SUM vm; pos = corpus.positions(b)
53    (wte, wpe, lm, wq, wk, wv, wo, f1, f2) = (view(p, 0), view(p, 1), view(p, 2), view(p, 3), view(p, 4), view(p, 5), view(p, 6), view(p, 7), view(p, 8))
54    h(m: Array[\RR64,(ZZ32,ZZ32)\]): Array[\RR64,(ZZ32,ZZ32,ZZ32)\] = heads(m, blockSize, nHead, headDim)
55    u(t: Array[\RR64,(ZZ32,ZZ32,ZZ32)\]): Array[\RR64,(ZZ32,ZZ32)\] = unheads(t, nHead)
56    (* -- forward, all (doc, pos) rows at once -- *)
57    x = gather(wte, ids) + gather(wpe, pos); xp = rows(rmsn, x); x1 = rows(rmsn, xp)
58    (q, k, v) = (x1 wq^T, x1 wk^T, x1 wv^T); (qh, kh, vh) = (h(q), h(k), h(v))
59    a = rows(sm, mask + (qh kh^T) / SQRT (1.0 headDim)); hc = u(a vh)
60    x2 = xp + hc wo^T; x3 = rows(rmsn, x2); m0 = x3 f1^T; mr = 0.0 MAX m0; x4 = x2 + mr f2^T
61    pr = rows(sm, x4 lm^T); loss = -(vm DOT log(pick(pr, tg))) / nv
62    (* -- backward, one line under each forward line, bottom-up -- *)
63    dL = diag(vm / nv) (pr - onehot(tg, vocabSize)); gLM = dL^T x4; dX4 = dL lm
64    gF2 = dX4^T mr; dM0 = (dX4 f2) × (m0 > 0.0); gF1 = dM0^T x3; dX3 = dM0 f1
65    dX2 = dX4 + rows(rmsn_b, dX3, x2); gWO = dX2^T hc; dh = h(dX2 wo)
66    dVh = a^T dh; dS = rows(sm_b, a, dh vh^T) / SQRT (1.0 headDim)
67    dQh = dS kh; dKh = dS^T qh; (dQ, dK, dV) = (u(dQh), u(dKh), u(dVh))
68    (gWQ, gWK, gWV) = (dQ^T x1, dK^T x1, dV^T x1); dX1 = dQ wq + dK wk + dV wv
69    dXp = dX2 + rows(rmsn_b, dX1, xp); dX = rows(rmsn_b, dXp, x)
70    gWTE = transpose(onehot(ids, vocabSize)) dX; gWPE = transpose(onehot(pos, blockSize)) dX
71    (loss, flat(gWTE, gWPE, gLM, gWQ, gWK, gWV, gWO, gF1, gF2))
72  end
73
74(* --------------------- Adam: three whole-vector expressions over p, m, v -- *)
75adam(p: Array[\RR64,ZZ32\], m: Array[\RR64,ZZ32\], v: Array[\RR64,ZZ32\], g: Array[\RR64,ZZ32\], t: ZZ32, lr: RR64):
76    (Array[\RR64,ZZ32\], Array[\RR64,ZZ32\], Array[\RR64,ZZ32\]) = do
77    m' = beta1 m + (1 - beta1) g
78    v' = beta2 v + (1 - beta2) (g × g)
79    p' = p - (lr (m' / (1 - beta1^t))) / (SQRT (v' / (1 - beta2^t)) + epsilon_A)
80    (p', m', v')
81  end
82learningRate(s: ZZ32): RR64 = lr0 (1 - (1.0 s) / nSteps)
83
84(* --------------------------------------------------- training ----------- *)
85run(): () = do
86    println "microgpt flat | params " nParams() " | corpus " corpus.size() " x " (blockSize + 1) " | batch 1, 5 steps"
87    bsz = 1
88    p: Array[\RR64,ZZ32\] := loadParams(weightsDir, matName, 9, nParams())
89    m: Array[\RR64,ZZ32\] := zeros(nParams())
90    v: Array[\RR64,ZZ32\] := zeros(nParams())
91    for s <- seq(0#5) do
92        t0 = nanoTime()
93        (l, g) = step(p, keys(bsz, fn (i: ZZ32): ZZ32 => (s bsz + i) MOD corpus.size()))
94        (p, m, v) := adam(p, m, v, g, s + 1, learningRate(s))
95        println "step " (s + 1) " loss " l "  (" ((nanoTime() - t0) DIV 1000000) " ms)"
96    end
97  end
98end
