(* q01b_wrapper: the elementwise algebra carried by a thin wrapper type of our
   own, form (ii) of the round-two comparison.  Every operator is a functional

    opr /(self, o: Arr[\I\]): Arr[\I\] = Arr[\I\](a.ivmap[\RR64\](fn (i: I, e: RR64): RR64 => e / o.a[i]))
    opr SQRT(self): Arr[\I\] = Arr[\I\](a.map[\RR64\](fn (e: RR64): RR64 => SQRT e))
    (* re-exports of what the bare array already had *)
    opr juxtaposition(self, s: RR64): Arr[\I\] = Arr[\I\](a.map[\RR64\](fn (e: RR64): RR64 => e s))

(* FlatData -- the loaders of Run C: a float parser (the library's strToFloat
   mishandles a sign and overflows on a long mantissa, ledger row 140), a
   whitespace tokenizer (String.split is a rope subdivision, row 141), text
   files as lines, vectors and matrices, and the corpus as one padded integer
   matrix with a length per document.  The check and the model share these. *)
component FlatData
export FlatData
import File.{...}

(* decimal digits, an optional '.', an optional e/E exponent; a leading '-' or '¯'.
   Digits accumulate in an RR64 mantissa, so a 17-digit mantissa is within one
   or two units in the last place of a correctly rounded parse, not exact. *)
parseFloat(s: String): RR64 = do
    neg = (s[0] = '-') OR (s[0] = '¯')
    m: RR64 := 0.0
    scale: ZZ32 := 0
    ex: ZZ32 := 0
    i: ZZ32 := if neg then 1 else 0 end
    seenPoint: Boolean := false
    n = |s|
    while (i < n) AND: ((s[i] =/= 'e') AND (s[i] =/= 'E')) do
        c = s[i]
        if c = '.' then seenPoint := true
        else
            m := 10.0 m + (c.codePoint - '0'.codePoint)
            if seenPoint then scale += 1 end
        end
        i += 1
    end
    if i < n then
        eneg = (s[i+1] = '-') OR (s[i+1] = '¯')
        j = if eneg OR (s[i+1] = '+') then i + 2 else i + 1 end
        ex := strToInt(s[j # (n - j)])
        if eneg then ex := -ex end
    end
    e10 = ex - scale
    r = if e10 >= 0 then m (10.0^e10)
        elif e10 >= -22 then m / (10.0^(-e10))
        else (m / (10.0^22)) / (10.0^(-e10 - 22)) end
    if neg then -r else r end
  end

(* the space-separated numbers of a line: one pass to count, one to parse *)
numbersOf(line: String): Array[\RR64,ZZ32\] = do
    n = |line|
    cnt: ZZ32 := 0
    i: ZZ32 := 0
    while i < n do
        if line[i] = ' ' then i += 1
        else
            cnt += 1
            while (i < n) AND: (line[i] =/= ' ') do i += 1 end
        end
    end
    out = array[\RR64\](cnt).fill(0.0)
    k: ZZ32 := 0
    i := 0
    while i < n do
        if line[i] = ' ' then i += 1
        else
            j: ZZ32 := i
            while (j < n) AND: (line[j] =/= ' ') do j += 1 end
            out[k] := parseFloat(line[i # (j - i)])
            k += 1
            i := j
        end
    end
    out
  end

(* the non-empty lines of a text file: one pass to count, one to read *)
readLines(path: String): Array[\String,ZZ32\] = do
    f = FileReadStream(path)
    cnt: ZZ32 := 0
    line: String := f.readLine()
    while |line| > 0 do cnt += 1; line := f.readLine() end
    f.close()
    g = FileReadStream(path)
    out = array[\String\](cnt).fill("")
    for i <- seq(0#cnt) do out[i] := g.readLine() end
    g.close()
    out
  end
(* one number per line *)
readVector(path: String, n: ZZ32): Array[\RR64,ZZ32\] = do
    f = FileReadStream(path)
    v = array[\RR64\](n).fill(0.0)
    for i <- seq(0#n) do v[i] := parseFloat(f.readLine()) end
    f.close()
    v
  end
(* one row per line, the numbers space-separated *)
readMatrix(path: String, rows: ZZ32, cols: ZZ32): Array[\RR64,(ZZ32,ZZ32)\] = do
    f = FileReadStream(path)
    m = array[\RR64\](rows, cols).fill(0.0)
    for i <- seq(0#rows) do
        nums = numbersOf(f.readLine())
        if |nums| =/= cols then fail("row " i " of " path " has " (|nums|) " numbers, expected " cols) end
        for j <- 0#cols do m[i, j] := nums[j] end
    end
    f.close()
    m
  end

(* the corpus as one integer matrix (documents x width): pad, the characters
   as 0..25, pad in every unused slot; and 1 + the name's length per document *)
loadCorpus(path: String, width: ZZ32, pad: ZZ32): (Array[\ZZ32,(ZZ32,ZZ32)\], Array[\ZZ32,ZZ32\]) = do
    names = readLines(path)
    nd = |names|
    tokm = array[\ZZ32\](nd, width).fill(pad)
    len = array[\ZZ32\](nd).fill(0)
    for d <- 0#nd do
        name = names[d]
        for j <- seq(0#|name|) do tokm[d, 1 + j] := name[j].codePoint - 'a'.codePoint end
        len[d] := 1 + |name|
    end
    (tokm, len)
  end
end

    ids = ravel(block(toks, 0, 0, bsz, blockSize)); tg = ravel(block(toks, 0, 1, bsz, blockSize))
    vm = ravel(valid(gather(corpusLengths, b), blockSize)); nv = SUM vm; pos = keys(n, fn (r: ZZ32): ZZ32 => r MOD blockSize)

    h(m: Array[\RR64,(ZZ32,ZZ32)\]): Array[\RR64,(ZZ32,ZZ32,ZZ32)\] = heads(m, blockSize, nHead, headDim)
    u(t: Array[\RR64,(ZZ32,ZZ32,ZZ32)\]): Array[\RR64,(ZZ32,ZZ32)\] = unheads(t, nHead)
    (* -- forward, all (doc, pos) rows at once -- *)
    x = gather(wte, ids) + gather(wpe, pos); xp = rows(rmsn, x); x1 = rows(rmsn, xp)
    (q, k, v) = (x1 (wq^T), x1 (wk^T), x1 (wv^T)); (qh, kh, vh) = (h(q), h(k), h(v))
    a = rows(sm, mask + (qh (kh^T)) / SQRT (1.0 headDim)); hc = u(a vh)
    x2 = xp + hc (wo^T); x3 = rows(rmsn, x2); m0 = x3 (f1^T); mr = 0.0 MAX m0; x4 = x2 + mr (f2^T)
    pr = rows(sm, x4 (lm^T)); loss = -(vm DOT log(pick(pr, tg))) / nv
    (* -- backward, one line under each forward line, bottom-up -- *)
    dL = (vm / nv) × (pr - onehot(tg, vocabSize)); gLM = dL^T x4; dX4 = dL lm
    gF2 = dX4^T mr; dM0 = (dX4 f2) × (m0 > 0.0); gF1 = dM0^T x3; dX3 = dM0 f1
    dX2 = dX4 + rows(rmsn_b, dX3, x2); gWO = dX2^T hc; dh = h(dX2 wo)
    dVh = (a^T) dh; dS = rows(sm_b, a, dh (vh^T)) / SQRT (1.0 headDim)
    dQh = dS kh; dKh = (dS^T) qh; (dQ, dK, dV) = (u(dQh), u(dKh), u(dVh))
    (gWQ, gWK, gWV) = (dQ^T x1, dK^T x1, dV^T x1); dX1 = dQ wq + dK wk + dV wv
    dXp = dX2 + rows(rmsn_b, dX1, xp); dX = rows(rmsn_b, dXp, x)
    gWTE = transpose(onehot(ids, vocabSize)) dX; gWPE = transpose(onehot(pos, blockSize)) dX
    (loss, flat(gWTE, gWPE, gLM, gWQ, gWK, gWV, gWO, gF1, gF2))
  end

(* --------------------- Adam: three whole-vector expressions over p, m, v -- *)
adam(p: Array[\RR64,ZZ32\], m: Array[\RR64,ZZ32\], v: Array[\RR64,ZZ32\], g: Array[\RR64,ZZ32\], t: ZZ32, lr: RR64):
    (Array[\RR64,ZZ32\], Array[\RR64,ZZ32\], Array[\RR64,ZZ32\]) = do
    m' = beta1 m + (1 - beta1) g
    v' = beta2 v + (1 - beta2) (g × g)
    p' = p - (lr (m' / (1 - beta1^t))) / (SQRT (v' / (1 - beta2^t)) + epsilon_A)
    (p', m', v')
  end
learningRate(s: ZZ32): RR64 = lr0 (1 - (1.0 s) / nSteps)

(* --------------------------------------------------- training ----------- *)
run(): () = do
    println "microgpt flat | params " nParams() " | corpus " nDocs() " x " (blockSize + 1) " | batch 1, 5 steps"
    bsz = 1
    p: Array[\RR64,ZZ32\] := loadParams(weightsDir)
    m: Array[\RR64,ZZ32\] := zeros(nParams())
    v: Array[\RR64,ZZ32\] := zeros(nParams())
    for s <- seq(0#5) do
        t0 = nanoTime()
        (l, g) = step(p, keys(bsz, fn (i: ZZ32): ZZ32 => (s bsz + i) MOD nDocs()))
        (p, m, v) := adam(p, m, v, g, s + 1, learningRate(s))
        println "step " (s + 1) " loss " l "  (" ((nanoTime() - t0) DIV 1000000) " ms)"
    end
  end
end

def dedent(lines):
    ind = min(len(l) - len(l.lstrip()) for l in lines if l.strip())
    return [l[ind:] for l in lines]

def split_stmts(lines):
    # for display, a `; ` line of the source is shown one statement per line (a do ... end on one line stays)
    out = []
    for l in lines:
        if '; ' in l and ' = do ' not in l:
            parts = l.split('; ')
            out.append(parts[0])
            out.extend(parts[1:])
        else:
            out.append(l)
    return out
