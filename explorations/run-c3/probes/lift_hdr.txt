object Row[\nat r, nat c\](base: Matrix[\RR64,r,c\], i: ZZ32) extends Vector[\RR64,c\]
    get(j: ZZ32): RR64 = base.get(i, j)
    put(j: ZZ32, v: RR64): () = base.put((i, j), v)
    init0(j: ZZ32, v: RR64): () = base.init0((i, j), v)
    replica[\U\](): Array1[\U,0,c\] = array1[\U,c\]()
end
rowOf[\nat r, nat c\](m: Matrix[\RR64,r,c\], i: ZZ32): Vector[\RR64,c\] = Row[\r,c\](m, i)

vec(n: ZZ32, f: ZZ32 -> RR64): Array[\RR64,ZZ32\] = array[\RR64\](n).fill(f)
mat(r: ZZ32, c: ZZ32, f: (ZZ32,ZZ32) -> RR64): Array[\RR64,(ZZ32,ZZ32)\] = array[\RR64\](r,c).fill(f)

opr /[\nat n\](v: Vector[\RR64,n\], s: RR64): Array[\RR64,ZZ32\] = vec(|v|, fn (j: ZZ32): RR64 => v[j] / s)
opr -[\nat n\](v: Vector[\RR64,n\], s: RR64): Array[\RR64,ZZ32\] = vec(|v|, fn (j: ZZ32): RR64 => v[j] - s)
opr SQRT[\nat n\](v: Vector[\RR64,n\]): Array[\RR64,ZZ32\] = vec(|v|, fn (j: ZZ32): RR64 => SQRT (v[j]))
opr ×[\nat n, nat m\](a: Vector[\RR64,n\], b: Vector[\RR64,m\]): Array[\RR64,ZZ32\] = vec(|a|, fn (j: ZZ32): RR64 => a[j] b[j])
exp[\nat n\](v: Vector[\RR64,n\]): Array[\RR64,ZZ32\] = vec(|v|, fn (j: ZZ32): RR64 => exp(v[j]))

epsRms: RR64 = 1.0 / (10.0^5)

rmsn(a: Array[\RR64,ZZ32\]): Array[\RR64,ZZ32\] = a / SQRT (epsRms + (a DOT a) / (1.0 |a|))
sm(z: Array[\RR64,ZZ32\]): Array[\RR64,ZZ32\] = do
    e = exp(z - (BIG MAX[u <- z] u))
    e / (SUM[u <- e] u)
  end
rmsnB(g: Array[\RR64,ZZ32\], a: Array[\RR64,ZZ32\]): Array[\RR64,ZZ32\] = do
    r = SQRT (epsRms + (a DOT a) / (1.0 |a|))
    y = a / r
    (g - ((y DOT g) / (1.0 |y|)) y) / r
  end
sumOf[\nat r, nat c\](m: Matrix[\RR64,r,c\]): RR64 = do
    (rr, cc) = m.sizes
    SUM[i <- 0#rr, j <- 0#cc] m[i,j]
  end
sumV[\nat n\](v: Vector[\RR64,n\]): RR64 = SUM[j <- 0#|v|] v[j]
ms(t0: ZZ64, t1: ZZ64): RR64 = (1.0 (t1 - t0)) / 1000000.0
